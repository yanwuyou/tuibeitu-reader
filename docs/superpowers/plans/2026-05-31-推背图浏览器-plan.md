### Task 1: 从 Markdown 提取六十象数据

**Files:**
- Create: `E:\推背图浏览器\scripts\extract_data.py`
- Read: `E:\焱无忧知识库\02-领域\历史\推背图-金圣叹批注版-全文.md`

**目标:** 写一个 Python 脚本，解析 markdown 文件，输出六十象的结构化 JSON。

- [ ] **Step 1: 编写提取脚本**

```python
import re, json

with open(r"E:\焱无忧知识库\02-领域\历史\推背图-金圣叹批注版-全文.md", encoding="utf-8") as f:
    text = f.read()

# 按 --- 分割各象
sections = re.split(r'\n---\n', text)
images = []

for sec in sections:
    m = re.search(r'## 第(.+?)象 (.+?)\n', sec)
    if not m:
        continue
    gan_zhi_title = m.group(2).strip()
    gm = re.match(r'(.+?)（(.+?)）', gan_zhi_title)
    gan_zhi = gm.group(1) if gm else gan_zhi_title
    title = gm.group(2) if gm else ""

    chen = ""
    song = ""
    jinpi = ""

    cm = re.search(r'\*\*谶曰：?\*\*\s*\n(.+?)(?:\n\n|\*\*颂|$)', sec, re.DOTALL)
    if cm: chen = cm.group(1).strip()

    sm = re.search(r'\*\*颂曰：?\*\*\s*\n(.+?)(?:\n\n|\*\*金|$)', sec, re.DOTALL)
    if sm: song = sm.group(1).strip()

    jm = re.search(r'\*\*金圣叹注解：?\*\*\s*(.+?)$', sec, re.DOTALL)
    if jm: jinpi = jm.group(1).strip()

    images.append({
        "id": len(images) + 1,
        "ganZhi": gan_zhi,
        "title": title,
        "chen": chen,
        "song": song,
        "jinPi": jinpi,
        "baiHua": ""  # Task 2 填入
    })

print(f"解析到 {len(images)} 象")

with open(r"E:\推背图浏览器\data_raw.json", "w", encoding="utf-8") as f:
    json.dump(images, f, ensure_ascii=False, indent=2)
print("已写入 data_raw.json")
```

  { name: "总纲", range: [1, 1] },

let currentIndex = 0; // 0-based
let currentTab = "chen";

function getCurrentImage() {
  return TUI_BEI_DATA[currentIndex];
}

function getCurrentDynasty() {
  const id = currentIndex + 1;
  return DYNASTIES.find(d => id >= d.range[0] && id <= d.range[1]);
}

- [ ] **Step 2: 渲染时间线**

```javascript
function renderTimeline() {
  const nav = document.getElementById("timeline");
  const activeDynasty = getCurrentDynasty();
  nav.innerHTML = DYNASTIES.map(d => {
    const isActive = d === activeDynasty;
    return `
      <div class="timeline-node${isActive ? " active" : ""}"
           data-start="${d.range[0]}"
           title="${d.name}（第${d.range[0]}-${d.range[1]}象）">
        <div class="dot"></div>
        <span>${d.name}</span>
        <span class="range">${d.range[0]}-${d.range[1]}象</span>
      </div>
    `;
  }).join("");

  nav.querySelectorAll(".timeline-node").forEach(node => {
    node.addEventListener("click", () => {
      const start = parseInt(node.dataset.start);
      navigateTo(start - 1);
    });
  });
}
```

- [ ] **Step 3: 渲染卡片内容**

```javascript
function updateTabContent() {
  const img = getCurrentImage();
  const content = document.getElementById("tabContent");
  if (currentTab === "chen") content.textContent = img.chen;
  else if (currentTab === "song") content.textContent = img.song;
  else content.textContent = img.jinPi;

  document.querySelectorAll(".tab").forEach(t => {
    t.classList.toggle("active", t.dataset.tab === currentTab);
  });
}

function renderCard() {
  const img = getCurrentImage();

  updateTabContent();
  document.getElementById("baihuaContent").textContent = img.baiHua || "（解读待补充）";
  document.getElementById("indicator").textContent =
    `第 ${img.id} / 60 象 · ${img.ganZhi}`;

  document.getElementById("prevBtn").disabled = currentIndex === 0;
  document.getElementById("nextBtn").disabled = currentIndex === 59;

  renderTimeline();
  window.location.hash = img.id;
}

// Tab 切换事件
document.addEventListener("DOMContentLoaded", () => {
  document.querySelectorAll(".tab").forEach(tab => {
    tab.addEventListener("click", () => {
      currentTab = tab.dataset.tab;
      updateTabContent();
    });
  });
});
```

- [ ] **Step 4: 翻页与键盘导航**

```javascript
function navigateTo(index) {
  if (index < 0 || index >= TUI_BEI_DATA.length) return;
  currentIndex = index;
  renderCard();
  window.scrollTo({ top: 0, behavior: "smooth" });
}

document.addEventListener("DOMContentLoaded", () => {
  document.getElementById("prevBtn").addEventListener("click", () => {
    navigateTo(currentIndex - 1);
  });
  document.getElementById("nextBtn").addEventListener("click", () => {
    navigateTo(currentIndex + 1);
  });
});

document.addEventListener("keydown", (e) => {
  if (e.key === "ArrowLeft") navigateTo(currentIndex - 1);
  if (e.key === "ArrowRight") navigateTo(currentIndex + 1);
});
```
